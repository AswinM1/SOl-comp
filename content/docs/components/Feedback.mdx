# Feedback
<div className="max-w-3xl min-h-[400px]  flex max-h-2xl p-4 justify-center mx-0  items-center bg-neutral-900 rounded-md">
<Feedback/>
</div>

A compact "Feedback" pill that smoothly expands into a textarea form on click, then collapses back after submit or cancel. The container itself animates its size and border radius with a spring, while its contents cross-fade between the collapsed button and the expanded form.

## Features

- **Morphing container** — a single `motion.div` animates `width`, `height`, and `borderRadius` between a small pill (95×42) and a large panel (420×300) using a spring transition.
- **Cross-faded content swap** — `AnimatePresence mode="popLayout"` swaps between the collapsed button and the expanded form with independent enter/exit fade + scale animations.
- **Controlled textarea** — the feedback text is held in local state and passed to `onSubmit` on confirm, then cleared automatically.
- **Auto-reset on close** — both submitting and cancelling clear the textarea and collapse the panel, so the form always reopens empty.
- **Fully configurable copy** — title, placeholder, and button labels are all overridable via props, with sensible defaults.
- **Autofocus** — the textarea receives focus automatically as soon as the form appears, so users can start typing immediately.

## Installation

This component depends on `motion` (the successor package to `framer-motion`) and Tailwind CSS for styling.

```bash
npm install motion
```

Make sure Tailwind CSS is configured in your project, then copy the component below into your codebase, e.g. `components/feedback.tsx`.

## Usage

```tsx
import Feedback from "@/components/feedback";

export default function Page() {
  return (
    <Feedback
      title="Got thoughts?"
      placeholder="What's on your mind?"
      onSubmit={(feedback) => {
        console.log("Feedback submitted:", feedback);
      }}
    />
  );
}
```

### Starting open by default

```tsx
<Feedback defaultOpen />
```

## Component

```tsx
"use client";

import { useState } from "react";
import { motion, AnimatePresence } from "motion/react";

interface FeedbackProps {
  title?: string;
  placeholder?: string;
  submitText?: string;
  cancelText?: string;
  onSubmit?: (feedback: string) => void;
  defaultOpen?: boolean;
}

function Feedback({
  title = "Feedback",
  placeholder = "Tell us what you think...",
  submitText = "Submit",
  cancelText = "Cancel",
  onSubmit,
  defaultOpen = false,
}: FeedbackProps) {
  const [open, setOpen] = useState<boolean>(defaultOpen);
  const [feedback, setFeedback] = useState<string>("");

  const handleSubmit = (): void => {
    onSubmit?.(feedback);
    setFeedback("");
    setOpen(false);
  };

  const handleCancel = (): void => {
    setFeedback("");
    setOpen(false);
  };

  return (
    <div className="mx-auto flex items-center justify-center">
      <AnimatePresence>
        <motion.div
          layout
          initial={{
            width: 95,
            height: 42,
            borderRadius: 6,
          }}
          animate={{
            width: open ? 420 : 95,
            height: open ? 300 : 42,
            borderRadius: open ? 18 : 6,
          }}
          transition={{
            type: "spring",
            stiffness: 180,
            damping: 22,
            mass: 0.8,
          }}
          className="
            overflow-hidden
            border border-neutral-600
            bg-neutral-900
            font-sans
            tracking-tight
          "
        >
          <AnimatePresence mode="popLayout">
            {!open ? (
              <motion.button
                key="button"
                type="button"
                onClick={() => setOpen(true)}
                className="
                  h-full w-full
                  cursor-pointer
                  text-sm font-medium
                "
                initial={{ opacity: 0 }}
                animate={{ opacity: 1 }}
                exit={{ opacity: 0 }}
              >
                {title}
              </motion.button>
            ) : (
              <motion.div
                key="form"
                className="flex h-full w-full flex-col p-4"
                initial={{
                  opacity: 0,
                  scale: 0.92,
                }}
                animate={{
                  opacity: 1,
                  scale: 1,
                }}
                exit={{
                  opacity: 0,
                  scale: 0.92,
                }}
                transition={{
                  duration: 0.25,
                  delay: 0.08,
                }}
              >
                <textarea
                  autoFocus
                  value={feedback}
                  onChange={(event) =>
                    setFeedback(event.target.value)
                  }
                  placeholder={placeholder}
                  className="
                    min-h-0 flex-1
                    resize-none
                    rounded-xl
                    border border-neutral-600
                    bg-neutral-300
                    p-4
                    text-sm text-black
                    outline-none
                    placeholder:text-neutral-500
                    focus:border-neutral-400
                  "
                />

                <div className="mt-3 flex justify-end gap-2">
                  <button
                    type="button"
                    onClick={handleCancel}
                    className="
                      cursor-pointer
                      rounded-lg
                      border border-neutral-600
                      px-4 py-2
                      text-sm
                    "
                  >
                    {cancelText}
                  </button>

                  <button
                    type="button"
                    onClick={handleSubmit}
                    className="
                      cursor-pointer
                      rounded-lg
                      bg-white
                      px-4 py-2
                      text-sm text-black
                    "
                  >
                    {submitText}
                  </button>
                </div>
              </motion.div>
            )}
          </AnimatePresence>
        </motion.div>
      </AnimatePresence>
    </div>
  );
}

export default Feedback;
```

## Props

| Prop | Type | Default | Description |
|---|---|---|---|
| `title` | `string` | `"Feedback"` | Label shown on the collapsed pill button. |
| `placeholder` | `string` | `"Tell us what you think..."` | Placeholder text inside the textarea once expanded. |
| `submitText` | `string` | `"Submit"` | Label for the confirm button. |
| `cancelText` | `string` | `"Cancel"` | Label for the cancel button. |
| `onSubmit` | `(feedback: string) => void` | `undefined` | Called with the current textarea value when the user clicks submit. Fired *before* the local `feedback` state is cleared. |
| `defaultOpen` | `boolean` | `false` | Whether the panel starts expanded on mount instead of collapsed. |
