# Action
<div className="max-w-3xl flex min-h-[400px] max-h-xl p-2 justify-center mx-0 items-center bg-neutral-900 rounded-md">
<Action/>
</div>


A single interactive element that starts as a small pill button, expands into a code-entry form on click, then automatically cycles through a "Processing…" spinner and a "Done" checkmark state before collapsing back to its original button — all via one shared `layoutId` so the shape morphs smoothly between every stage.

## Features

- **Shared layout morphing** — the collapsed button and the expanded panel share a `layoutId="action"`, so Framer Motion automatically animates the size/position/shape change between them instead of a plain cross-fade.
- **Multi-stage state machine** — moves through `"input"` → `"processing"` → `"done"` → back to `"input"` (and closed), each stage rendering different content with its own enter/exit animation.
- **Blur-based transitions** — every stage transition animates `filter: blur()` alongside opacity, giving a soft, camera-focus-like feel rather than a flat fade.
- **Configurable timing** — how long the processing and done states are shown for is controlled via props, not hardcoded.
- **Controlled input with validation** — the textarea's value is held in state and passed through to `onComplete`; "Done" is disabled until non-whitespace text is entered.
- **Cancelable mid-flow** — a "Cancel" button in the input stage clears any pending timers and resets the widget immediately.
- **Completion callback** — `onComplete(code)` fires once the whole flow finishes and the widget has reset, receiving the code the user entered.
- **Auto-reset** — after showing "Done," the component automatically closes and resets its internal state back to `"input"`, ready to be reopened.
- **Leak-safe timers** — pending `setTimeout`s are tracked in a ref and cleared on unmount or cancel, so no stray state updates fire after the widget is gone.

## Installation

This component depends on `motion` (the successor package to `framer-motion`), `lucide-react` for icons, and Tailwind CSS for styling.

```bash
npm install motion lucide-react
```
```bash
npx shadcn@latest add https://sol-components.vercel.app/r/action.json
```

Make sure Tailwind CSS is configured in your project, then copy the component below into your codebase, e.g. `components/action.tsx`.

## Usage

```tsx
import Action from "@/components/action";

export default function Page() {
  return (
    <Action
      placeholder="Enter your invite code"
      processingDuration={1500}
      doneDuration={1200}
      onComplete={() => {
        console.log("Code redeemed!");
      }}
    />
  );
}
```



## Component

```tsx
"use client";

import React, { useState, useRef, useEffect } from "react";
import { motion, AnimatePresence, type Transition } from "motion/react";
import { LoaderCircle, Check } from "lucide-react";

type ActionStatus = "input" | "processing" | "done";

interface ActionProps {
  placeholder?: string;
  processingDuration?: number;
  doneDuration?: number;
  onComplete?: (code: string) => void;
}

const spring: Transition = {
  type: "spring",
  stiffness: 260,
  damping: 26,
  mass: 0.9,
};

function Action({
  placeholder = "Enter your code",
  processingDuration = 2000,
  doneDuration = 1000,
  onComplete,
}: ActionProps) {
  const [open, setOpen] = useState<boolean>(false);
  const [status, setStatus] = useState<ActionStatus>("input");
  const [code, setCode] = useState<string>("");

  const timeoutsRef = useRef<ReturnType<typeof setTimeout>[]>([]);

  const clearPendingTimeouts = (): void => {
    timeoutsRef.current.forEach(clearTimeout);
    timeoutsRef.current = [];
  };

  useEffect(() => {
    return () => {
      clearPendingTimeouts();
    };
  }, []);

  const handleOpen = (): void => {
    setOpen(true);
  };

  const handleDone = (): void => {
    if (!code.trim()) return;

    setStatus("processing");

    const processingTimeout = setTimeout(() => {
      setStatus("done");

      const doneTimeout = setTimeout(() => {
        setOpen(false);
        setStatus("input");
        onComplete?.(code);
        setCode("");
      }, doneDuration);

      timeoutsRef.current.push(doneTimeout);
    }, processingDuration);

    timeoutsRef.current.push(processingTimeout);
  };

  const handleCancel = (): void => {
    clearPendingTimeouts();
    setOpen(false);
    setStatus("input");
    setCode("");
  };

  return (
    <div className="flex items-center justify-center">
      <AnimatePresence mode="popLayout" initial={false}>
        {!open ? (
          <motion.button
            key="button"
            layoutId="action"
            layout
            onClick={handleOpen}
            style={{ borderRadius: 10 }}
            className="
              h-fit w-fit p-3
              cursor-pointer
              border border-neutral-600
              bg-black
               text-sm font-medium tracking-tight
              text-neutral-200
            "
            transition={{ layout: spring }}
          >
            <motion.span
              initial={{ opacity: 0, filter: "blur(4px)" }}
              animate={{ opacity: 1, filter: "blur(0px)" }}
              transition={{ duration: 0.2 }}
            >
              I have a code
            </motion.span>
          </motion.button>
        ) : (
          <motion.div
            key="content"
            layoutId="action"
            layout
            style={{ borderRadius: 14 }}
            className="
              relative z-50
              flex w-fit items-center
              overflow-hidden
              border border-neutral-600
              bg-black
              p-3
               tracking-tight
            "
            transition={{ layout: spring }}
          >
            <AnimatePresence mode="popLayout" initial={false}>
              {status === "input" && (
                <motion.div
                  key="input"
                  layout
                  className="flex items-center gap-3"
                  initial={{
                    opacity: 0,
                    filter: "blur(8px)",
                    x: -8,
                  }}
                  animate={{
                    opacity: 1,
                    filter: "blur(0px)",
                    x: 0,
                  }}
                  exit={{
                    opacity: 0,
                    filter: "blur(8px)",
                    x: -8,
                  }}
                  transition={{ duration: 0.25 }}
                >
                  <textarea
                    autoFocus
                    value={code}
                    onChange={(event) => setCode(event.target.value)}
                    placeholder={placeholder}
                    className="
                      h-12 w-64
                      resize-none overflow-hidden
                      rounded-md
                      border border-neutral-700
                      bg-neutral-950
                      p-3
                      text-neutral-100
                      placeholder:text-neutral-500
                      outline-none
                    "
                  />

                  <button
                    type="button"
                    onClick={handleCancel}
                    className="
                      w-fit cursor-pointer
                      rounded-xl
                      border border-neutral-700
                      p-3
                      font-medium
                      text-neutral-300
                    "
                  >
                    Cancel
                  </button>

                  <button
                    type="button"
                    onClick={handleDone}
                    disabled={!code.trim()}
                    className="
                      w-fit cursor-pointer
                      rounded-xl
                      bg-white
                      p-3
                      font-medium
                      text-black
                      disabled:cursor-not-allowed
                      disabled:opacity-40
                    "
                  >
                    Done
                  </button>
                </motion.div>
              )}

              {status === "processing" && (
                <motion.div
                  key="processing"
                  layout
                  className="
                    flex items-center gap-3
                    px-3 py-2
                    text-sm text-neutral-200
                  "
                  initial={{
                    opacity: 0,
                    filter: "blur(8px)",
                    scale: 0.95,
                  }}
                  animate={{
                    opacity: 1,
                    filter: "blur(0px)",
                    scale: 1,
                  }}
                  exit={{
                    opacity: 0,
                    filter: "blur(8px)",
                    scale: 0.95,
                  }}
                  transition={{ duration: 0.25 }}
                >
                  <LoaderCircle size={18} className="animate-spin" />
                  <span>Processing...</span>
                </motion.div>
              )}

              {status === "done" && (
                <motion.div
                  key="done"
                  layout
                  className="
                    flex items-center gap-3
                    px-3 py-2
                    text-sm text-neutral-200
                  "
                  initial={{
                    opacity: 0,
                    filter: "blur(8px)",
                    scale: 0.95,
                  }}
                  animate={{
                    opacity: 1,
                    filter: "blur(0px)",
                    scale: 1,
                  }}
                  transition={{ duration: 0.3 }}
                >
                  <Check size={18} />
                  <span>Done</span>
                </motion.div>
              )}
            </AnimatePresence>
          </motion.div>
        )}
      </AnimatePresence>
    </div>
  );
}

export default Action;
```

## Props

| Prop | Type | Default | Description |
|---|---|---|---|
| `placeholder` | `string` | `"Enter your code"` | Placeholder text shown in the textarea during the `"input"` stage. |
| `processingDuration` | `number` | `2000` | Milliseconds the `"processing"` (spinner) stage is shown before advancing to `"done"`. |
| `doneDuration` | `number` | `1000` | Milliseconds the `"done"` (checkmark) stage is shown before the widget auto-closes and resets. |
| `onComplete` | `(code: string) => void` | `undefined` | Called once the flow fully finishes, with the code the user entered. Fires after the widget has already closed and reset to `"input"`. |
